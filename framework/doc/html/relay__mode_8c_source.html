115static void load_relay_mmwlan_ap_args(struct mmwlan_ap_args *ap_args)
116{
117 char strval[32];
118 uint32_t uint32val;
119
120 /* Load BSSID */
121 if (mmconfig_read_string("wlan.ap_bssid", strval, sizeof(strval)) > 0)
122 {
123 int temp[6];
124 int i;
125
126 int ret = sscanf(strval,
127 "%x:%x:%x:%x:%x:%x",
128 &temp[0],
129 &temp[1],
130 &temp[2],
131 &temp[3],
132 &temp[4],
133 &temp[5]);
134 if (ret == 6)
135 {
136 for (i = 0; i < 6; i++)
137 {
138 if (temp[i] > UINT8_MAX || temp[i] < 0)
139 {
140 /* Invalid value, ignore and reset to default */
141 memset(ap_args->bssid, 0, sizeof(ap_args->bssid));
142 break;
143 }
144
145 ap_args->bssid[i] = (uint8_t)temp[i];
146 }
147 }
148 }
149
150 /* Load DTIM period */
151 if (mmconfig_read_uint32("wlan.ap_dtim_period", &uint32val) == MMCONFIG_OK)
152 {
153 if (uint32val <= UINT16_MAX)
154 {
155 ap_args->dtim_period = uint32val;
156 }
157 else
158 {
159 printf("%s out of range\n", "wlan.ap_dtim_period");
160 }
161 }
162
163 /* Load Beacon Interval */
164 if (mmconfig_read_uint32("wlan.ap_beacon_interval_tus", &uint32val) == MMCONFIG_OK)
165 {
166 if (uint32val <= UINT16_MAX)
167 {
168 ap_args->beacon_interval_tus = uint32val;
169 }
170 else
171 {
172 printf("%s out of range\n", "wlan.ap_beacon_interval_tus");
173 }
174 }

180static void load_mmwlan_relay_settings(void)
181{
182 /* Optionally override the relay depth. Only valid while STA and AP are both disabled, so it
183 * must be applied before connecting to the network. */
184 uint32_t relay_depth_override;
185 if (mmconfig_read_uint32("wlan.relay_depth_override", &relay_depth_override) == MMCONFIG_OK)
186 {
187 if (relay_depth_override <= MMWLAN_RELAY_MAX_DEPTH)
188 {
189 enum mmwlan_status status =
190 mmwlan_relay_set_depth_override((uint8_t)relay_depth_override);
191 if (status != MMWLAN_SUCCESS)
192 {
193 MMLOG_ERR("Failed to set relay depth override (status %d)\n", status);
194 }
195 }
196 else
197 {
198 printf("%s out of range\n", "wlan.relay_depth_override");
199 }
200 }

207void app_init(void)
208{
209 enum mmwlan_status status;
210
211 MMLOG_PRINTF("\n\nRelay Mode Example (Built " __DATE__ " " __TIME__ ")\n\n");
212
213 relay_led_init();
214
215 /* Initialise the Wi-Fi system and boot the chip. */
216 app_wlan_init();
217
218 load_mmwlan_relay_settings();
219
220 /* Bring up the relay subsystem. */
221 struct mmwlan_relay_args relay_args = MMWLAN_RELAY_ARGS_INIT;
222 status = mmwlan_relay_enable(&relay_args);
223 if (status != MMWLAN_SUCCESS)
224 {
225 MMLOG_ERR("Failed to start relay (status %d)\n", status);
226 relay_led_set_failed();
227 MMOSAL_ASSERT(0);
228 }
229
230 status = mmwlan_register_depth_change_cb(handle_relay_depth_change, &opaque_argument_value);
231 if (status != MMWLAN_SUCCESS)
232 {
233 MMLOG_ERR("Failed to register depth change callback (status %d)\n", status);
234 }
235
236 /* Connect to Wi-Fi network, block till connected. */
237 relay_led_set_connecting();
238 app_wlan_start();
239 if (mmwlan_get_sta_state() == MMWLAN_STA_CONNECTED)
240 {
241 relay_led_set_connected();
242 }
243 else
244 {
245 relay_led_set_failed();
246 }
247 mmwlan_set_power_save_mode(MMWLAN_PS_DISABLED);
248
249 /* Load the sta args shared by the whole relay tree. */
250 struct mmwlan_sta_args sta_args = MMWLAN_STA_ARGS_INIT;
251 load_mmwlan_sta_args(&sta_args);
252
253 /* Bring up the AP interface. */
254 struct mmwlan_ap_args ap_args = MMWLAN_AP_ARGS_INIT;
255 load_relay_mmwlan_ap_args(&ap_args);
256
257 /* The AP re-advertises the STA credentials so the whole tree shares one SSID. */
258 memcpy(ap_args.ssid, sta_args.ssid, sizeof(ap_args.ssid));
259 ap_args.ssid_len = sta_args.ssid_len;
260 ap_args.security_type = sta_args.security_type;
261 memcpy(ap_args.passphrase, sta_args.passphrase, sizeof(ap_args.passphrase));
262 ap_args.passphrase_len = sta_args.passphrase_len;
263 ap_args.pmf_mode = sta_args.pmf_mode;
264
265 ap_args.sta_status_cb = handle_ap_sta_status;
266 ap_args.sta_status_cb_arg = &opaque_argument_value;
267
268 ap_args.max_stas = MAX_STAS;
269
270 status = mmwlan_ap_enable(&ap_args);
271 if (status == MMWLAN_SUCCESS)
272 {
273 MMLOG_PRINTF("AP started with SSID: %s ", ap_args.ssid);
274 if (ap_args.security_type == MMWLAN_SAE)
275 {
276 MMLOG_PRINTF("with passphrase %s", ap_args.passphrase);
277 }
278 MMLOG_PRINTF("\n");
279 }
280 else
281 {
282 MMLOG_PRINTF("Failed to start AP (status %d)\n", status);
283 relay_led_set_failed();
284 }
285}
