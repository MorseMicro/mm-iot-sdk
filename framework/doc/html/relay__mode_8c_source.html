50void app_init(void)
51{
52 enum mmwlan_status status;
53
54 MMLOG_PRINTF("\n\nRelay Mode Example (Built " __DATE__ " " __TIME__ ")\n\n");
55
56 struct mmwlan_relay_status last_status = {
57 .state = MMWLAN_RELAY_STATE_DISABLED,
58 .sta_state = MMWLAN_STA_DISABLED,
59 .depth = MMWLAN_RELAY_DEPTH_UNKNOWN,
60 };
61 struct relay_update node_state = {
62 .status = last_status,
63 .semb = mmosal_semb_create("node_state"),
64 };
65 MMOSAL_ASSERT(node_state.semb != NULL);
66
67 relay_led_init();
68
69 /* Initialise the Wi-Fi system and boot the chip. */
70 app_wlan_init();
71
72 status = mmwlan_register_relay_status_cb(handle_relay_status_update, &node_state);
73 MMOSAL_ASSERT(status == MMWLAN_SUCCESS);
74
75 /* Bring up the relay subsystem. */
76 relay_led_set_connecting();
77 app_wlan_start_relay(true);
78 relay_led_set_connected();
79
80 MMLOG_APP("Relay started successfully\n");
81
82 while (mmosal_semb_wait(node_state.semb, UINT32_MAX))
83 {
84 last_status = node_state.status;
85 relay_led_set_depth(last_status.depth);
86
87 uint8_t parent_bssid[MMWLAN_MAC_ADDR_LEN] = { 0 };
88 char parent_str[30] = { 0 };
89 char *state_str = "UNKNOWN";
90 switch (last_status.sta_state)
91 {
92 case MMWLAN_STA_DISABLED:
93 state_str = "disabled";
94 break;
95
96 case MMWLAN_STA_CONNECTING:
97 state_str = "connecting";
98 break;
99
100 case MMWLAN_STA_CONNECTED:
101 state_str = "connected";
102 mmwlan_get_bssid(parent_bssid);
103 snprintf(parent_str,
104 sizeof(parent_str),
105 " to " MM_MAC_ADDR_FMT,
106 MM_MAC_ADDR_VAL(parent_bssid));
107 break;
108 }
109
110 char *role_str = "UNKNOWN";
111 switch (last_status.state)
112 {
113 case MMWLAN_RELAY_STATE_DISABLED:
114 role_str = "disabled";
115 break;
116
117 case MMWLAN_RELAY_STATE_STA:
118 role_str = "station";
119 break;
120
121 case MMWLAN_RELAY_STATE_REPEATER:
122 role_str = "repeater";
123 break;
124
125 case MMWLAN_RELAY_STATE_ROOT:
126 role_str = "root";
127 MMOSAL_ASSERT(0);
128 break;
129 }
130
131 char depth_num[4];
132 char *depth_str = "detached";
133 if (last_status.depth != MMWLAN_RELAY_DEPTH_UNKNOWN)
134 {
135 snprintf(depth_num, sizeof(depth_num), "%u", last_status.depth);
136 depth_str = depth_num;
137 }
138 MMLOG_APP("UPDATE - Relay STA: %s%s, Relay role: %s, Depth: %s\n",
139 state_str,
140 parent_str,
141 role_str,
142 depth_str);
143 }
144
145 mmosal_semb_delete(node_state.semb);
146}
