50static void led_task_main(void *arg)
51{
52 MM_UNUSED(arg);
53
54 while (true)
55 {
56 /* Clear LED for ease of state management */
57 mmhal_set_led(LED_RED, LED_OFF);
58 mmhal_set_led(LED_GREEN, LED_OFF);
59 mmhal_set_led(LED_BLUE, LED_OFF);
60
61 switch (led_mode)
62 {
63 case RELAY_LED_MODE_CONNECTING:
64 mmhal_set_led(LED_BLUE, LED_ON);
65 mmosal_task_sleep(CONNECTING_BLINK_PERIOD_MS);
66 mmhal_set_led(LED_BLUE, LED_OFF);
67 mmosal_task_sleep(CONNECTING_BLINK_PERIOD_MS);
68 break;
69
70 case RELAY_LED_MODE_CONNECTED:
71 if (led_depth == MMWLAN_RELAY_DEPTH_UNKNOWN)
72 {
73 if (mmwlan_get_sta_state() != MMWLAN_STA_CONNECTED)
74 {
75 relay_led_set_connecting();
76 break;
77 }
78 mmhal_set_led(LED_BLUE, LED_ON);
79 mmosal_task_sleep(LED_TASK_IDLE_POLL_MS);
80 }
81 else if (led_depth == 0)
82 {
83 mmhal_set_led(LED_GREEN, LED_ON);
84 mmosal_task_sleep(LED_TASK_IDLE_POLL_MS);
85 }
86 else
87 {
88 uint8_t depth = led_depth;
89
90 for (size_t i = 0; i < depth && led_mode == RELAY_LED_MODE_CONNECTED; i++)
91 {
92 mmhal_set_led(LED_GREEN, LED_ON);
93 mmosal_task_sleep(DEPTH_PULSE_MS);
94 mmhal_set_led(LED_GREEN, LED_OFF);
95 mmosal_task_sleep(DEPTH_PULSE_MS);
96 }
97 mmosal_task_sleep(DEPTH_PULSE_PAUSE_MS);
98 }
99 break;
100
101 case RELAY_LED_MODE_FAILED:
102 default:
103 mmhal_set_led(LED_RED, LED_ON);
104 mmosal_task_sleep(LED_TASK_IDLE_POLL_MS);
105 break;
106 }
107 }
108}

110void relay_led_init(void)
111{
112 MMOSAL_ASSERT(led_task == NULL);
113 led_task = mmosal_task_create(led_task_main, NULL, MMOSAL_TASK_PRI_LOW, 128, "relay_led");
114 MMOSAL_ASSERT(led_task != NULL);
115}
