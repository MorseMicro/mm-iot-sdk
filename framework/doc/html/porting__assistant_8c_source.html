100static const char *result_code_to_string(enum test_result result)
101{
102 switch (result)
103 {
104 case TEST_NO_RESULT:
105 return "";
106 case TEST_PASSED:
107 return F_GREEN("PASS");
108 case TEST_SKIPPED:
109 return F_BLUE("SKIP");
110 case TEST_FAILED:
111 return F_RED("FAIL");
112 case TEST_FAILED_NON_CRITICAL:
113 return F_YELLOW("FAIL");
114 }
115 MMOSAL_ASSERT(false);
116}

126static void run_test_steps(const struct test_step *const steps[],
127 size_t num_steps,
128 struct test_counters *ctrs)
129{
130 static char log_buf[1024];
131
132 size_t ii;
133 for (ii = 0; ii < num_steps; ii++)
134 {
135 const struct test_step *step = steps[ii];
136 size_t log_buf_len = sizeof(log_buf);
137 log_buf[0] = '\0';
138
139 LOG_PRINTF(F_BOLD("%-60s "), step->description);
140 LOG_FLUSH();
141
142 enum test_result result = step->exec(log_buf, log_buf_len);
143 if (result != TEST_NO_RESULT)
144 {
145 LOG_PRINTF("[ %s ]", result_code_to_string(result));
146 }
147 LOG_WRITE("\n");
148
149 if (log_buf[0] != '\0')
150 {
151 LOG_WRITE("\n");
152 LOG_WRITE(log_buf);
153 }
154 LOG_FLUSH();
155 switch (result)
156 {
157 case TEST_NO_RESULT:
158 ctrs->no_result++;
159 break;
160 case TEST_PASSED:
161 ctrs->pass++;
162 break;
163 case TEST_SKIPPED:
164 break;
165 case TEST_FAILED:
166 ctrs->fail++;
167 break;
168 case TEST_FAILED_NON_CRITICAL:
169 ctrs->fail++;
170 break;
171 }
172
173 if (result == TEST_FAILED)
174 {
175 break;
176 }
177 }
178}

184void app_init(void)
185{
186 /* Having deep sleep enabled can complicate debugging. Given the purpose of this applications
187 * is to validate that you can communicate to the MM-Chip it will be disabled by default. */
188 mmhal_set_deep_sleep_veto(MMHAL_VETO_ID_APP_MIN);
189
190 struct test_counters ctrs = { 0 };
191 unsigned num_tests = sizeof(test_steps) / sizeof(test_steps[0]);
192
193 char hw_version_string[32] = { 0 };
194
195 bool ok = mmhal_get_hardware_version(hw_version_string, sizeof(hw_version_string));
196 if (!ok)
197 {
198 snprintf(hw_version_string, sizeof(hw_version_string), "%s", "Unknown");
199 }
200
201 LOG_WRITE(F_BOLD("\n\nMM-IoT-SDK Porting Assistant\n"));
202 LOG_WRITE("----------------------------\n");
203 LOG_PRINTF(" HW Version: %s\n", hw_version_string);
204 LOG_PRINTF(" Morselib version: %s\n", MM_VERSION_BUILDID);
205 LOG_WRITE("----------------------------\n\n");
206
207 run_test_steps(test_steps, num_tests, &ctrs);
208
209 LOG_PRINTF("\n\n%u total test steps. %u passed, %u failed, %u no result, %u skipped\n",
210 num_tests,
211 ctrs.pass,
212 ctrs.fail,
213 ctrs.no_result,
214 num_tests - ctrs.no_result - ctrs.pass - ctrs.fail);
215}
