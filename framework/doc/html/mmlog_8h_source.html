242#if MMLOG_LEVEL >= MMLOG_LEVEL_ERR
244#define MMLOG(fmt, _col, _lvl, ...) \
245 do { \
246 char lvl = (_lvl); \
247 char col = (_col); \
248 char short_task_name[3] = { '?', '?', '\0' }; \
249 const char *task_name = mmosal_task_name(); \
250 if (task_name != NULL) \
251 { \
252 short_task_name[0] = task_name[0]; \
253 short_task_name[1] = task_name[1]; \
254 } \
255 (void)(lvl); \
256 (void)(col); \
257 (void)(short_task_name); \
258 MMLOG_PRINTF(MMLOG_PREFIX_FMT fmt MMLOG_SUFFIX_FMT, MMLOG_PREFIX_ARGS, ##__VA_ARGS__); \
259 } while (0)
