216static inline const struct mmwlan_s1g_channel_list *mmwlan_lookup_regulatory_domain(
217 const struct mmwlan_regulatory_db *db,
218 const char *country_code)
219{
220 unsigned ii;
221
222 if (db == NULL)
223 {
224 return NULL;
225 }
226
227 for (ii = 0; ii < db->num_domains; ii++)
228 {
229 const struct mmwlan_s1g_channel_list *channel_list = db->domains[ii];
230 if (channel_list->country_code[0] == country_code[0] &&
231 channel_list->country_code[1] == country_code[1])
232 {
233 return channel_list;
234 }
235 }
236 return NULL;
237}
