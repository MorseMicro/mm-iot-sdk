143static inline char mm_nibble_to_hex_char(uint8_t nibble)
144{
145 nibble &= 0x0f;
146 if (nibble < 0x0a)
147 {
148 return '0' + nibble;
149 }
150 else
151 {
152 return 'A' + nibble - 0x0a;
153 }
154}
155
163static inline int8_t mm_decode_hex_char(char c)
164{
165 if (c >= '0' && c <= '9')
166 {
167 return c - '0';
168 }
169 if (c >= 'a' && c <= 'f')
170 {
171 return c - 'a' + 0xa;
172 }
173 if (c >= 'A' && c <= 'F')
174 {
175 return c - 'A' + 0xa;
176 }
177 return -1;
178}

502static inline bool mm_str_to_mac_addr(const char *str, uint8_t *val)
503{
504 if (strnlen(str, MM_MAC_ADDR_STR_LEN) != MM_MAC_ADDR_STR_LEN - 1)
505 {
506 return false;
507 }
508
509 uint8_t temp[MMWLAN_MAC_ADDR_LEN];
510 for (int i = 0; i < MMWLAN_MAC_ADDR_LEN; ++i)
511 {
512 int hi = mm_decode_hex_char(str[0]);
513 int lo = mm_decode_hex_char(str[1]);
514 char mac_delim = str[2];
515 str += 3;
516 if (hi < 0 || lo < 0 || (mac_delim != ':' && mac_delim != '\0'))
517 {
518 return false;
519 }
520 temp[i] = (hi << 4) | lo;
521 }
522
523 memcpy(val, temp, sizeof(temp));
524 return true;
525}
