32static char *mmwlan_ap_sta_state_to_str(enum mmwlan_ap_sta_state state)
33{
34 switch (state)
35 {
36 case MMWLAN_AP_STA_UNKNOWN:
37 return "Unknown (disconnected)";
38
39 case MMWLAN_AP_STA_AUTHENTICATED:
40 return "Authenticated";
41
42 case MMWLAN_AP_STA_ASSOCIATED:
43 return "Associated";
44
45 case MMWLAN_AP_STA_AUTHORIZED:
46 return "Authorized";
47 }
48
49 return "Unrecognized";
50}

58static void handle_ap_sta_status(const struct mmwlan_ap_sta_status *sta_status, void *arg)
59{
60 /* Validate that the opaque argument received matches the value passed in. This is just for
61 * testing purposes. */
62 MMOSAL_ASSERT(arg == &opaque_argument_value);
63
64 MMLOG_APP("AP STA " MM_MAC_ADDR_FMT " State: %s, AID: %u\n",
65 MM_MAC_ADDR_VAL(sta_status->mac_addr),
66 mmwlan_ap_sta_state_to_str(sta_status->state),
67 sta_status->aid);
68}

75static void link_status_callback(const struct mmipal_link_status *link_status)
76{
77 uint32_t time_ms = mmosal_get_time_ms();
78 if (link_status->link_state == MMIPAL_LINK_UP)
79 {
80 MMLOG_PRINTF("Link is up. Time: %lu ms", time_ms);
81 MMLOG_PRINTF(", IP: %s", link_status->ip_addr);
82 MMLOG_PRINTF(", Netmask: %s", link_status->netmask);
83 MMLOG_PRINTF(", Gateway: %s\n", link_status->gateway);
84 }
85 else
86 {
87 MMLOG_PRINTF("Link is down. Time: %lu ms\n", time_ms);
88 }
89}

95void app_init(void)
96{
97 MMLOG_PRINTF("\n\nAP Mode Example (Built " __DATE__ " " __TIME__ ")\n\n");
98 mmwlan_init();
99 mmwlan_set_channel_list(load_channel_list());
100 mmwlan_boot(NULL);
101
102 app_print_version_info();
103
104 /* Load IP stack settings from config store, or use defaults if no entry found in
105 * config store. */
106 struct mmipal_init_args mmipal_init_args = MMIPAL_INIT_ARGS_DEFAULT;
107 load_ap_mmipal_init_args(&mmipal_init_args);
108
109 /* Initialize IP stack. */
110 if (mmipal_init(&mmipal_init_args) != MMIPAL_SUCCESS)
111 {
112 MMLOG_APP("Error initializing network interface.\n");
113 MMOSAL_ASSERT(false);
114 }
115
116 mmipal_set_link_status_callback(link_status_callback);
117
118 mmwlan_set_power_save_mode(MMWLAN_PS_DISABLED);
119 load_mmwlan_ap_settings();
120
121 struct mmwlan_ap_args ap_args = MMWLAN_AP_ARGS_INIT;
122 load_mmwlan_ap_args(&ap_args);
123
124 ap_args.sta_status_cb = handle_ap_sta_status;
125 ap_args.sta_status_cb_arg = &opaque_argument_value;
126
127 MMLOG_APP("Starting AP mode...\n");
128 enum mmwlan_status status = mmwlan_ap_enable(&ap_args);
129 if (status == MMWLAN_SUCCESS)
130 {
131 MMLOG_APP("AP started successfully\n");
132 }
133 else
134 {
135 MMLOG_APP("Failed to start AP (status %d)\n", status);
136 }
137 // do led flashing for num stas connected?
138}
